 55    def search_obj_by_stix_id(self, stix_id: str) -> dict:
 56        """
 57        Search the database for a STIX object with the provided STIX ID. 
 58        If the object is found, it retrieves its common properties.
 59
 60        :param stix_id: The STIX ID of the object to search for.
 61        :type stix_id: str
 62        :return: A dictionary containing the common properties of the found object,
 63                 or an empty dictionary if no object is found.
 64        :rtype: dict
 65        """
 66        query = f"{cons.MATCH}$so has stix-id '{stix_id}';" "get $so;"
 67        stix_obj = self.db_manager.get_query(query)
 68        if not stix_obj:
 69            return {}
 70        properties = self._get_common_properties_of(stix_obj[0].get("so"))
 71        return properties

 73    def search_obj_by_attck_id(self, mitre_attck_id: str, ignore_revoked=True) -> list[dict]:
 74        """
 75        Provide information of STIX objects associated to a MITRE ATT&CK ID.
 76        Only STIX objects of the type corresponding to the format of the given ID
 77        (as per the ATT&CK-stix2.1 specification) are selected. E.g., SDOs 'attack-pattern'
 78        for IDs of the form "T####".
 79
 80        :param mitre_attck_id: The MITRE ATT&CK ID to search for.
 81        :type mitre_attck_id: str
 82        :raises ValueError: If the format of the given MITRE ATT&CK ID is invalid.
 83
 84        :returns: A list of dictionaries, each containing the properties of a related STIX object.
 85        :rtype: list
 86        """
 87        res_var = "stix_obj"
 88        stix_type = CTIEngine._get_stix_type_from_id(mitre_attck_id)
 89
 90        query = cons.build_match_clause(
 91            cons.SEARCH_BY_ATTACK_ID.format(
 92            mitre_attck_id=mitre_attck_id, var=res_var)
 93        )
 94        if stix_type is not None:
 95            # att&ck ids that can be mapped to different stix types (e.g. software)
 96            if isinstance(stix_type, list):
 97                query += " or ".join([f"{{${res_var} isa {st};}}" for st in stix_type]) + ";\n"
 98            else:
 99                query += f"${res_var} isa {stix_type};\n"
100        if ignore_revoked:
101            query += f"${res_var} {cons.NOT_REVOKED};\n"
102        query += f"get ${res_var};"
103        logger.debug("Search query:\n%s" %query)
104        
105        stix_objs = self.db_manager.get_query(query)
106        if not stix_objs:
107            return []
108
109        related = []
110        for so in stix_objs:
111            properties = self._get_common_properties_of(so.get(res_var))
112            related.append(properties)
113        return related

115    def search_obj_by_name_alias(self, name_alias:str) -> list[Mnemonic]:

130                    name = r.get('n').get("value"),
131                    aliases = aliases
132                )
133            )

196    def get_techniques_used_by(self, group_ids: list[str], inference=False) -> dict:
197        """
198        Retrieve techniques found to be used by any of the members of a given set of groups,
199        along with the number of groups that use it.
200
201        :param group_id: A list of identifiers of ATT&CK groups.
202        :type group_id: list
203        :param inference: True if inference should be used in the query; defaults to False.

207        """
208        ids = "|".join(group_ids)
209        query = cons.build_match_clause(
210            cons.TECHNIQUES_USED_BY.format(group_ids=ids))
211        query += (
212            "get $id, $group;\n"
213            "group $id;\n"
214            "count;\n"
215        )
216        logger.debug("Techniques by any group:\n%s" %query)
217        data = self.db_manager.aggregate_group_query(query, inference)
218        return data

220    def get_techniques_at_intersection(self, group_ids: list[str], inference=False) -> dict:
221        """
222        Retrieve techniques found to be used by all the members of a given set of groups.
223
224        :param group_ids: A list of stix identifiers of the groups.
225        :type group_ids: list
226        :param inference: True if inference should be used in the query; defaults to False.
227        :type inference: bool, optional
228        :return: A dictionary with pairs [ATT&CK id:name] of the techniques
229        :rtype: dict
230        """
231        # we use 'replace' instead of 'format' to avoid conflicts due to
232        # the presence of {} in the TECHNIQUE_ID_FORMAT inner string
233        pattern = cons.TECHNIQUES_AS_STIX.replace(
234            "{revoked}", f"$t {cons.NOT_REVOKED};\n")
235        query = cons.build_match_clause(pattern)
236        for i, group_id in enumerate(group_ids):
237            query += f"$g{i} isa intrusion-set, has stix-id '{group_id}';\n"
238            query += f"(used: $t, user: $g{i}) isa uses;\n"
239        query += (
240            f"get {cons.XREF_ID_VAR}, $t_name;\n"
241            "sort $t_name;"
242        )
243        logger.debug("Techniques by all groups:\n%s" %query)
244        result = self.db_manager.get_query(query, inference)
245        return TypeDBQueryHandler.dict_from_answers(result, "eid", "t_name")

247    def explain_techniques_used_by(
248        self, group_mitre_ids: list[str], technique_stix_id: str = None
249    ) -> InferredAnswer:
250        """
251        Provides explanations tracing the inference rules that are applied to derive
252        the set of techniques used by the specified groups. The explanations can be optionally
253        filtered by the stix ID of a specific technique.
254
255        :param group_mitre_ids: The MITRE ATT&CK identifiers of the groups for which to explain techniques.
256        :type group_mitre_ids: list[str]
257        :param technique_stix_id: The stix ID of a specific technique to explain (optional).
258        :type technique_stix_id: str, optional
259        :return: An InferredAnswer object containing the explanation of the techniques
260            (or the selected technique) used by the group(s).
261        :rtype: InferredAnswer
262        """
263        ids = "|".join(group_mitre_ids)
264        if technique_stix_id is not None:
265            # the stix-id clause goes at the beginning to optimize the query evaluation
266            query = cons.build_match_clause(f"$technique has stix-id '{technique_stix_id}';")
267            query += cons.TECHNIQUES_USED_BY.format(group_ids=ids)
268        else:
269            query = cons.build_match_clause(cons.TECHNIQUES_USED_BY.format(group_ids=ids))
270        query += (
271            # add the technique name to the result set
272            f"\n$technique has name $technique_name;\n"
273            "get;\n")
274        return InferredAnswer(query, self.db_manager.explain_get_query(query))

276    def get_intrusion_sets_per_technique(
277        self,
278        ignore_revoked=True,
279        inference=False,
280    ) -> dict:
281        """
282        Retrieve the number of intrusion sets associated with MITRE ATT&CK techniques.
283
284        :param ignore_revoked: True (default) to ignore revoked techniques, False otherwise.
285        :type ignore_revoked: bool
286        :param inference: Flag to enable inference in the query. Default is False.
287        :type inference: bool
288
289        :returns: A dictionary where the keys are the ATT&CK IDs of the techniques,
290            and the value is the number of intrusion sets that use them.
291        :rtype: dict
292        """
293        revoke_flag = f"$t {cons.NOT_REVOKED};\n" if ignore_revoked else ""
294        pattern = cons.TECHNIQUES_AS_STIX.replace("{revoked}", revoke_flag)
295        query = cons.build_match_clause(pattern)
296        # groups using the techniques, aggregated by technique
297        query += (
298            "$group isa intrusion-set, has name $group_name;\n"
299            "$use (used: $t, user: $group) isa uses;\n"
300            f"get $t_name,{cons.XREF_ID_VAR},$group_name;\n"
301            f"group {cons.XREF_ID_VAR};\n"
302            "count;\n"
303        )
304        logger.debug("Intrusion sets per technique:\n%s" %query)
305        data = self.db_manager.aggregate_group_query(query, inference)
306        return data

309        """
310        Retrieves the names of techniques based on a list of ATT&CK IDs.
311
312        :param attck_ids: A list of MITRE ATT&CK IDs.
313        :type attck_ids: list of str

318        query = f"{cons.MATCH}{cons.XREF_IS_MITRE_ATTCK}{cons.XREF_HAS_ID}{cons.XREF_ID_VAR};"

320            [f"\n{{{cons.XREF_ID_VAR} == '{tid}';}}" for tid in attck_ids]
321        )
322        query = query.rstrip(" or") + ";\n"
323        query += (
324            "$sdo isa stix-domain-object, has name $sdo_name;\n"
325            f"$rel (referrer: $sdo, referenced: {cons.XREF_VAR})"
326            f"{cons.EXT_REFERENCING_REL}\n"
327            f"get {cons.XREF_ID_VAR}, $sdo_name;"
328        )
329        result = self.db_manager.get_query(query)
330        return TypeDBQueryHandler.dict_from_answers(result, "eid", "sdo_name")

347        """
348        Obtains a collection of courses of action that can be associated with a 
349         specific group via the "related-to" relation.

351         that otherwise would never be satisfied.
352
353        If both are given, the group_mitre_id is used. 
354        """

356            raise ValueError("Either group_mitre_id or group_name must be provided.")

359        else:
360            name = group_name
361        mit = self.db_manager.get_query(
362            cons.MITIGATIONS_REL_TECHNIQUE.format(name), inference=True
363        )
364        logger.debug(cons.MITIGATIONS_REL_TECHNIQUE.format(name))
365        return TypeDBQueryHandler.dict_from_answers(mit, "sid", "mitigation-name")

445    def filter_groups_keywords(self, keywords=None, all=True) -> list[Group]:
446        """
447        List all the MITRE ATT&CK groups
448        """
449        stix_type = cons.ATTCK_TO_STIX.get("group")
450        constraints = ""
451        if keywords:
452            if all:
453                constraint_lst = [f"has description contains '{kw}'" for kw in keywords]
454                constraints = f"${stix_type} {', '.join(constraint_lst)};\n"
455            else:
456                constraint_lst = [f"{{${stix_type} has description contains '{kw}';}}" for kw in keywords]
457                constraints = " or ".join(constraint_lst) + ";\n"
458
459        query = (
460            f"{cons.MATCH}{cons.XREF_IS_MITRE_ATTCK}"
461            f"{cons.XREF_HAS_ID}like '{cons.GROUP_ID_FORMAT}';\n"
462            f"{cons.JOIN_STIX_ATTCK.format(stix_type=stix_type, filters=constraints)}\n"
463            "fetch\n"
464            f"{cons.XREF_ID_VAR};\n"
465            f"${stix_type}:name,description,alias;\n"
466        )
467
468        logger.debug("Get groups by keywords:\n%s" %query)
469        result = self.db_manager.fetch_query(query)
470        res_as_list = []
471        for r in result:
472            al = r.get(stix_type).get("alias")
473            aliases = ", ".join([a.get("value") for a in al]) if al else ""
474            res_as_list.append(
475                Group(
476                    group_id = r.get('eid').get("value"),
477                    name = r.get(stix_type).get("name")[0].get("value"),
478                    description = r.get(stix_type).get("description")[0].get("value"),
479                    aliases = aliases
480                )
481            )
482        return res_as_list

486    def _get_stix_type_from_id(attck_id: str) -> str | list | None:
487        attck_id_patterns = {
488            cons.CAMPAIGN_ID_FORMAT: cons.ATTCK_TO_STIX.get("campaign"),
489            cons.GROUP_ID_FORMAT: cons.ATTCK_TO_STIX.get("group"),
490            cons.MITIGATION_ID_FORMAT: cons.ATTCK_TO_STIX.get("mitigation"),
491            cons.SOFTWARE_ID_FORMAT: cons.ATTCK_TO_STIX.get("software"),
492            cons.SUBTECHNIQUE_ID_FORMAT: cons.ATTCK_TO_STIX.get("subtechnique"),
493            cons.TECHNIQUE_ID_FORMAT: cons.ATTCK_TO_STIX.get("technique"),
494        }
495
496        for id_format, stix_type in attck_id_patterns.items():
497            if match(id_format, attck_id):
498                return stix_type
499        raise ValueError(
500            f"Invalid ATT&CK ID format in '{attck_id}'. "
501            "Valid formats are: 'C####', 'G####', 'M####', 'S####', 'T####[.###]'"
