 36class Thing:
 37    """Represents a Thing object in TypeDB, i.e. entity or relation.
 38    
 39    Note: The Thing object will be deprecated in TypeDB 3.0.
 40    """
 41
 42    def __init__(
 43            self,
 44            variable: str = "",
 45            typedb_type: str = typedb_constants.THING,
 46        ):
 47        """Instantiate a thing.
 48        
 49        :param variable: The name of the variable assigned to this 
 50            thing, defaults to an unused variable
 51        :type variable: str, optional
 52        :param typedb_type: The type in typedb of this thing, 
 53            defaults to "thing"
 54        :type typedb_type: str, optional
 55        """
 56        # Note: Using a default value in the constructor with a method 
 57        #       call leads to variables being skipped. 
 58        #       Therefore, "variable" is checked here.
 59        if variable == "":
 60            variable = VariableDealer.get_variable()
 61        self.variable: str = variable
 62        self.type: str = typedb_type
 63        # Note: For every attribute, a list of values is assigned.
 64        self.attributes: dict[str, list[str]] = {}
 65    
 66    def add_attribute(self, name: str, value: str) -> None:
 67        """Add an attribute to the thing.
 68        
 69        :param name: The name of the attribute
 70        :type name: str
 71        :param value: The value of the attribute
 72        :type value: str
 73        """
 74        if name and value:
 75            if not self.attributes.get(name):
 76                self.attributes[name] = []
 77            self.attributes[name].append(value)
 78
 79    def get_attributes(self) -> dict[str, list[str]]:
 80        """Returns the names of the attributes and the values that are
 81        assigned to them.
 82        
 83        :return: The names and values assigned to them
 84        :rtype: dict[str, list[str]]
 85        """
 86        return self.attributes
 87
 88    def set_type(self, object_type: str) -> None:
 89        """Set the type of the thing.
 90        
 91        :param object_type: The type of this thing
 92        :type object_type: str
 93        """
 94        self.type = object_type
 95
 96    def get_type(self) -> str:
 97        """Returns the type of the Thing.
 98        
 99        :return: The type of this thing
100        :rtype: str
101        """
102        return self.type
103
104    def get_variable(self) -> str:
105        """Returns the variable of the thing in the insert query.
106        
107        :return: The variable assigned to this Thing
108        :rtype: str
109        """
110        return self.variable
111
112    def __str__(self):
113        return f"{self.variable}:{self.type}"

143class Relation(Thing):
144    """Represents a Relation object in TypeDB."""
145
146    def __init__(
147            self,
148            variable: str = "", 
149            typedb_type: str = typedb_constants.RELATION, 
150        ):
151        """Instantiate a relation.
152        
153        :param variable: The name of the variable assigned to this 
154            relation, defaults to an unused variable
155        :type variable: str, optional
156        :param typedb_type: The type in typedb of this relation, 
157            defaults to "relation"
158        :type typedb_type: str, optional
159        """
160        super().__init__(variable, typedb_type)
161        # Note: For every role, a variable is assigned.
162        #       So the dict is in {role: [variable]} format
163        #       as in TypeQL queries.
164        self.plays_roles: dict[str, list[str]] = {}
165
166    def __eq__(self, other: Self) -> bool:
167        # Note: Variables are not relevant for equivalence
168        if isinstance(other, Relation):
169            return (other.get_type() == self.get_type()
170                    and other.get_attributes() == self.get_attributes()
171                    and other.get_roles().keys() == self.get_roles().keys())
172        return False
173
174    def add_roleplayer(self, role: str, variable: str) -> None:
175        """Add a role to the relation.
176        
177        :param role: The name of the role
178        :type role: str
179        :param variable: The variable of the entity that plays the role
180        :type variable: str
181        """
182        if self.plays_roles.get(role):
183            self.plays_roles[role].append(variable)
184        else:
185            self.plays_roles.update({role: [variable]})
186
187    def get_roles(self) -> dict[str, list[str]]:
188        """Returns the names and variables of the roles.
189        
190        :return: The names and variables of the roles
191        :rtype: dict[str, list[str]]
192        """
193        return self.plays_roles
194
195    def __str__(self):
196        return f"{super().__str__()},roles: {self.plays_roles}"

199class InsertQuery:
200    """Structure containing a set of TypeDb types/Things (entity, relation, attribute) that can be 
201    mapped to an Insert query in TypeQL."""
202
203    def __init__(self):
204        """Empty Insert query structure"""
205        self.match_clause: list[Thing] = []
206        self.insert_clause: list[Thing] = []
207
208    def add_to_match_clause(self, instance: Thing) -> None:
209        """Add an instance of a type (or object) to the match clause.
210
211        :param instance: The instance to add
212        :type instance: Thing
213        """
214        self.match_clause.append(instance)
215
216    def add_to_insert_clause(self, insert: Thing) -> None:
217        """Add a thing to the insert clause.
218
219        :param insert: The thing to add
220        :type insert: Thing
221        """
222        self.insert_clause.append(insert)
223
224    def get_match_clause(self) -> list[Thing]:
225        """Returns the things in the match clause.
226
227        :return: The things in the match clause
228        :rtype: list[Thing]
229        """
230        return self.match_clause
231
232    def get_insert_clause(self) -> list[Thing]:
233        """Returns the Things in the insert clause.
234
235        :return: The things in the insert clause
236        :rtype: list[Thing]
237        """
238        return self.insert_clause
239
240    def is_empty(self) -> bool:
241        """States whether this InsertQuery is empty, i.e. whether there
242        is no insertion.
243
244        :return: Whether there is no insertion
245        :rtype: bool
246        """
247        return len(self.insert_clause) == 0
248
249    def extend(self, insert_query: Self) -> None:
250        """Combines this InsertQuery with a given one by clauses
251
252        :param insert_query: The other InsertQuery
253        :type insert_query: InsertQuery
254        """
255        self.match_clause.extend(insert_query.get_match_clause())
256        self.insert_clause.extend(insert_query.get_insert_clause())
257
258    def __str__(self):
259        res = "match: ["
260        for e in self.match_clause:
261            res += e.__str__() + "; "
262        res += "]\ninsert: ["
263        for e in self.insert_clause:
264            res += e.__str__() + "; "
265        res += "]\n"
266
267        return res
